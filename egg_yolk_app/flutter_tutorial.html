🏆 predictImage() — ประกอบร่างสูตร SVR

local_predict_service.dartSVR RBF Kernel Evaluation
163    final normFeatures = List<double>.generate(6, (i) {
164      return (rawFeatures[i] - _scalerMean![i]) / _scalerScale![i];
165    });  // Standardize: (x - mean) / std เหมือน StandardScaler ของ Python
167    // f(x) = b + sum(alpha_i * exp(-gamma * ||x - sv_i||^2))
168    double rawScore = _intercept!;           // เริ่มที่ intercept (b)
172    for (int i = 0; i < numSVs; i++) {      // วน 556 Support Vectors
174      double distSq = 0.0;
175      for (int j = 0; j < 6; j++) {
176        final diff = normFeatures[j] - sv[j];
177        distSq += diff * diff;               // ||x - sv_i||^2
178      }
179      final rbf = math.exp(-gamma * distSq); // K = exp(-γd²)
180      rawScore += _dualCoef![i] * rbf;        // += α_i * K
181    }
184    final predictedScore = rawScore.round().clamp(1, 15);
SVR RBF Formula: f(x) = b + Σᵢ αᵢ · exp(-γ · ||x - sᵢ||²) · x = feature standardized, sᵢ = support vector i, αᵢ = dual coef, γ = gamma rawScore.round().clamp(1, 15) เทียบ Python: int(round(max(1.0, min(15.0, raw_score))))
